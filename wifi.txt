  Microcontroller: ESP8266 (ESP-01 / ESP-01S)
  Description:

// --- Send Full Schedule & Medicine Names to STM32 over UART ---
void sendScheduleToSTM32() {
  // Protocol: "SCHED:HH:MM,E,HH:MM,E,HH:MM,E,HH:MM,E\n"
  Serial.print("SCHED:");
  Serial.print(schedule.dose1); Serial.print(","); Serial.print(schedule.en1 ? "1" : "0"); Serial.print(",");
  Serial.print(schedule.dose2); Serial.print(","); Serial.print(schedule.en2 ? "1" : "0"); Serial.print(",");
  Serial.print(schedule.dose3); Serial.print(","); Serial.print(schedule.en3 ? "1" : "0"); Serial.print(",");
  Serial.print(schedule.dose4); Serial.print(","); Serial.println(schedule.en4 ? "1" : "0");

  delay(50);
  // Transmit Medicine Names
  Serial.print("NAME1:"); Serial.println(schedule.name1);
  delay(30);
  Serial.print("NAME2:"); Serial.println(schedule.name2);
  delay(30);
  Serial.print("NAME3:"); Serial.println(schedule.name3);
  delay(30);
  Serial.print("NAME4:"); Serial.println(schedule.name4);
}

void setupEEPROM() {
  EEPROM.begin(sizeof(Schedule) + 16);
  EEPROM.get(EEPROM_ADDR, schedule);

  // Validate or set defaults
  if (schedule.dose1[0] < '0' || schedule.dose1[0] > '2') {
    strcpy(schedule.dose1, "08:00");
    strcpy(schedule.dose2, "13:00");
    strcpy(schedule.dose3, "18:00");
    strcpy(schedule.dose4, "21:00");
    schedule.en1 = true;
    schedule.en2 = true;
    schedule.en3 = true;
    schedule.en4 = true;
    strcpy(schedule.name1, "Paracetamol");
    strcpy(schedule.name2, "Amoxicillin");
    strcpy(schedule.name3, "Vitamin C");
    strcpy(schedule.name4, "Melatonin");
    EEPROM.put(EEPROM_ADDR, schedule);
    EEPROM.commit();
  }
}

void setup() {
  // Serial baud rate for STM32 communication
  Serial.begin(115200);
  delay(500);

  setupEEPROM();

  // Configure Soft Access Point
  WiFi.mode(WIFI_AP);
  WiFi.softAP(apSSID, apPASS);

  // Server Routes
  server.on("/", HTTP_GET, []() {
    server.send_P(200, "text/html", INDEX_HTML);
  });

  server.on("/api/get", HTTP_GET, []() {
    String json = "{";
    json += "\"d1\":\"" + String(schedule.dose1) + "\",";
    json += "\"d2\":\"" + String(schedule.dose2) + "\",";
    json += "\"d3\":\"" + String(schedule.dose3) + "\",";
    json += "\"d4\":\"" + String(schedule.dose4) + "\",";
    json += "\"n1\":\"" + String(schedule.name1) + "\",";
    json += "\"n2\":\"" + String(schedule.name2) + "\",";
    json += "\"n3\":\"" + String(schedule.name3) + "\",";
    json += "\"n4\":\"" + String(schedule.name4) + "\",";
    json += "\"e1\":" + String(schedule.en1 ? "true" : "false") + ",";
    json += "\"e2\":" + String(schedule.en2 ? "true" : "false") + ",";
    json += "\"e3\":" + String(schedule.en3 ? "true" : "false") + ",";
    json += "\"e4\":" + String(schedule.en4 ? "true" : "false");
    json += "}";
    server.send(200, "application/json", json);
  });

  server.on("/api/save", HTTP_POST, []() {
    if (server.hasArg("d1")) strncpy(schedule.dose1, server.arg("d1").c_str(), 5);
    if (server.hasArg("d2")) strncpy(schedule.dose2, server.arg("d2").c_str(), 5);
    if (server.hasArg("d3")) strncpy(schedule.dose3, server.arg("d3").c_str(), 5);
    if (server.hasArg("d4")) strncpy(schedule.dose4, server.arg("d4").c_str(), 5);
    schedule.dose1[5] = '\0';
    schedule.dose2[5] = '\0';
    schedule.dose3[5] = '\0';
    schedule.dose4[5] = '\0';

    if (server.hasArg("n1")) strncpy(schedule.name1, server.arg("n1").c_str(), 15);
    if (server.hasArg("n2")) strncpy(schedule.name2, server.arg("n2").c_str(), 15);
    if (server.hasArg("n3")) strncpy(schedule.name3, server.arg("n3").c_str(), 15);
    if (server.hasArg("n4")) strncpy(schedule.name4, server.arg("n4").c_str(), 15);
    schedule.name1[15] = '\0';
    schedule.name2[15] = '\0';
    schedule.name3[15] = '\0';
    schedule.name4[15] = '\0';

    schedule.en1 = (server.arg("e1") == "1");
    schedule.en2 = (server.arg("e2") == "1");
    schedule.en3 = (server.arg("e3") == "1");
    schedule.en4 = (server.arg("e4") == "1");

    EEPROM.put(EEPROM_ADDR, schedule);
    EEPROM.commit();

    sendScheduleToSTM32();
    server.send(200, "text/plain", "OK");
  });

  server.on("/api/synctime", HTTP_GET, []() {
    if (server.hasArg("time")) {
      String t = server.arg("time"); // e.g. "2026-10-14 08:30:00"
      Serial.print("TIME:");
      Serial.println(t);
      server.send(200, "text/plain", "OK");
    } else {
      server.send(400, "text/plain", "Missing time");
    }
  });

  server.on("/api/dispense", HTTP_GET, []() {
    Serial.println("CMD:DISPENSE");
    server.send(200, "text/plain", "OK");
  });

  server.begin();

  // Send initial schedule to STM32 upon power up
  delay(1000);
  sendScheduleToSTM32();
}

void loop() {
  server.handleClient();

  // Optional: Read any status reports sent back from STM32
  if (Serial.available()) {
    String msg = Serial.readStringUntil('\n');
    msg.trim();
  }
}
